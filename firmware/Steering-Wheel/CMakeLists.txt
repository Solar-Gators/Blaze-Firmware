cmake_minimum_required(VERSION 3.20)

project(Steering-Wheel LANGUAGES C CXX ASM)

# ============================================================
# 1. Include directories
# ============================================================

set(NODE_INCLUDES
        Core/Inc
        user/Inc

        Drivers/STM32U5xx_HAL_Driver/Inc
        Drivers/STM32U5xx_HAL_Driver/Inc/Legacy

        Drivers/CMSIS/Device/ST/STM32U5xx/Include
        Drivers/CMSIS/Include
        Drivers/CMSIS/RTOS2/Include

        # FreeRTOS: use ONLY the NTZ port (the one whose sources are compiled below).
        # Mixing ARM_CM33 and ARM_CM33_NTZ gives duplicate port symbols / wrong portmacro.h.
        Middlewares/Third_Party/FreeRTOS/Source/include
        Middlewares/Third_Party/FreeRTOS/Source/portable/GCC/ARM_CM33_NTZ/non_secure
        Middlewares/Third_Party/FreeRTOS/Source/CMSIS_RTOS_V2

        TouchGFX/App
        TouchGFX/gui/include
        TouchGFX/generated/gui_generated/include
        TouchGFX/generated/fonts/include
        TouchGFX/generated/images/include
        TouchGFX/generated/texts/include
        TouchGFX/target
        TouchGFX/target/generated

        # TouchGFX framework (folder is lowercase "touchgfx" as generated by Designer)
        Middlewares/ST/touchgfx/framework/include

        # GPU2D / NeoChrom with vector graphics (STM32U5G9): NemaPVG, Cortex-M33.
        # NemaGFX and TouchGFXNema must be the SAME variant.
        Middlewares/ST/touchgfx_components/gpu2d/TouchGFXNema_NemaPVG_m33_r01/include
        Middlewares/ST/touchgfx_components/gpu2d/NemaGFX_NemaPVG_m33_r01/include

        # Shared project resources
        ../../resources/inc
        ../../resources/Drivers
)

# ============================================================
# 2. Collect application and HAL sources
# ============================================================

file(GLOB_RECURSE CUBEMX_SOURCES CONFIGURE_DEPENDS
        "Core/Src/*.c"
        "Core/Src/*.cpp"
        "user/Src/*.c"
        "user/Src/*.cpp"
        "Drivers/STM32U5xx_HAL_Driver/Src/*.c"

        "Middlewares/Third_Party/FreeRTOS/Source/*.c"
        "Middlewares/Third_Party/FreeRTOS/Source/portable/GCC/ARM_CM33_NTZ/non_secure/*.c"
        "Middlewares/Third_Party/FreeRTOS/Source/portable/MemMang/*.c"
        "Middlewares/Third_Party/FreeRTOS/Source/CMSIS_RTOS_V2/*.c"

        "TouchGFX/App/*.c"
        "TouchGFX/App/*.cpp"
        "TouchGFX/gui/src/*.cpp"
        "TouchGFX/generated/*.c"
        "TouchGFX/generated/*.cpp"
        "TouchGFX/target/*.c"
        "TouchGFX/target/*.cpp"
)

# GLOB_RECURSE on "Middlewares/Third_Party/FreeRTOS/Source/*.c" also pulls in every
# portable/* port (IAR, other cores, ...). Keep only the NTZ port, MemMang and CMSIS-RTOS2.
set(_FILTERED "")
foreach(_src ${CUBEMX_SOURCES})
    if(_src MATCHES "FreeRTOS/Source/portable/")
        if(_src MATCHES "portable/GCC/ARM_CM33_NTZ/non_secure/" OR
                _src MATCHES "portable/MemMang/heap_1\\.c$")
            list(APPEND _FILTERED "${_src}")
        endif()
    else()
        list(APPEND _FILTERED "${_src}")
    endif()
endforeach()
set(CUBEMX_SOURCES ${_FILTERED})

# PC simulator code must never be built for the MCU.
list(FILTER CUBEMX_SOURCES EXCLUDE REGEX "TouchGFX/generated/simulator/")
list(FILTER CUBEMX_SOURCES EXCLUDE REGEX "TouchGFX/simulator/")

# Exclude HAL template files.
list(FILTER CUBEMX_SOURCES EXCLUDE REGEX ".*_template\\.c$")

list(REMOVE_DUPLICATES CUBEMX_SOURCES)

# ============================================================
# 3. Startup assembly
# ============================================================

set(STARTUP_SRC "${CMAKE_CURRENT_SOURCE_DIR}/CubeMX/startup_stm32u5g9xx.s")

if(NOT EXISTS "${STARTUP_SRC}")
    file(GLOB STARTUP_CANDIDATES CONFIGURE_DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/CubeMX/startup_*.s"
            "${CMAKE_CURRENT_SOURCE_DIR}/CubeMX/startup_*.S"
            "${CMAKE_CURRENT_SOURCE_DIR}/startup_*.s"
            "${CMAKE_CURRENT_SOURCE_DIR}/startup_*.S"
    )
    if(STARTUP_CANDIDATES)
        list(GET STARTUP_CANDIDATES 0 STARTUP_SRC)
    else()
        message(FATAL_ERROR "No STM32 startup assembly file found.")
    endif()
endif()

# ============================================================
# 4. Validate required sources and dependencies
# ============================================================

if(NOT CUBEMX_SOURCES)
    message(FATAL_ERROR "No application source files found.")
endif()

if(NOT TARGET SolarGators_drivers)
    message(FATAL_ERROR "SolarGators_drivers target must be defined before this CMakeLists.")
endif()

# TouchGFX prebuilt libraries (core + GPU2D binding + NemaGFX driver).
set(_ST "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/ST")
set(TOUCHGFX_LIB     "${_ST}/touchgfx/lib/core/cortex_m33/gcc/libtouchgfx-float-abi-hard.a")
set(TOUCHGFXNEMA_LIB "${_ST}/touchgfx_components/gpu2d/TouchGFXNema_NemaPVG_m33_r01/lib/cortex_m33/gcc/libtouchgfxnema-float-abi-hard.a")
set(NEMAGFX_LIB      "${_ST}/touchgfx_components/gpu2d/NemaGFX_NemaPVG_m33_r01/lib/core/cortex_m33/gcc/libnemagfx-float-abi-hard.a")

foreach(_lib TOUCHGFX_LIB TOUCHGFXNEMA_LIB NEMAGFX_LIB)
    if(NOT EXISTS "${${_lib}}")
        message(FATAL_ERROR "Missing library: ${${_lib}}")
    endif()
endforeach()

# ============================================================
# 5. Executable
# ============================================================

add_executable(${PROJECT_NAME}.elf
        ${CUBEMX_SOURCES}
        ${STARTUP_SRC}
)

target_include_directories(${PROJECT_NAME}.elf PRIVATE ${NODE_INCLUDES})

target_compile_definitions(${PROJECT_NAME}.elf PRIVATE
        STM32U5G9xx
        USE_HAL_DRIVER
)

target_compile_options(${PROJECT_NAME}.elf PRIVATE
        -mcpu=cortex-m33
        -mthumb
        -mfpu=fpv5-sp-d16
        -mfloat-abi=hard
        -fdata-sections
        -ffunction-sections
        -Wall
        -Wextra
        -Wshadow
        -Wdouble-promotion
        -Wformat=2
        # TouchGFX is built without RTTI/exceptions
        $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
        $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>
        $<$<COMPILE_LANGUAGE:CXX>:-fno-threadsafe-statics>
)

# ============================================================
# 6. Libraries
# ============================================================

target_compile_definitions(SolarGators_drivers PRIVATE
        STM32U5G9xx
        USE_HAL_DRIVER
)

target_include_directories(SolarGators_drivers PRIVATE ${NODE_INCLUDES})

target_link_libraries(${PROJECT_NAME}.elf PRIVATE
        SolarGators_drivers
        # Static libs reference each other; group them so link order does not matter.
        -Wl,--start-group
        "${TOUCHGFX_LIB}"
        "${TOUCHGFXNEMA_LIB}"
        "${NEMAGFX_LIB}"
        -Wl,--end-group
)

# ============================================================
# 7. Linker script
# ============================================================

# The CubeMX folder contains STM32U5G9xx_FLASH.ld and STM32U5G9xx_RAM.ld.
# Only ONE -T script may be passed.
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/CubeMX/STM32U5G9xx_FLASH.ld")

if(NOT EXISTS "${LINKER_SCRIPT}")
    file(GLOB LINKER_CANDIDATES CONFIGURE_DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/CubeMX/*FLASH*.ld"
            "${CMAKE_CURRENT_SOURCE_DIR}/*FLASH*.ld"
    )
    if(LINKER_CANDIDATES)
        list(GET LINKER_CANDIDATES 0 LINKER_SCRIPT)
    else()
        message(FATAL_ERROR "No FLASH linker script (.ld) found.")
    endif()
endif()

target_link_options(${PROJECT_NAME}.elf PRIVATE
        -mcpu=cortex-m33
        -mthumb
        -mfpu=fpv5-sp-d16
        -mfloat-abi=hard
        "-T${LINKER_SCRIPT}"
        "-Wl,-Map=${PROJECT_NAME}.map"
        -Wl,--gc-sections
        -Wl,--print-memory-usage
        --specs=nano.specs
        # NOTE: --specs=nosys.specs is already added globally by the root CMakeLists.txt.
        # Passing it twice makes GCC fail with "attempt to rename spec ...".
)

# ============================================================
# 8. Post-build artifacts
# ============================================================

add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O binary
        "$<TARGET_FILE:${PROJECT_NAME}.elf>"
        "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.bin"

        COMMAND ${CMAKE_OBJCOPY} -O ihex
        "$<TARGET_FILE:${PROJECT_NAME}.elf>"
        "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.hex"

        COMMAND ${CMAKE_SIZE} "$<TARGET_FILE:${PROJECT_NAME}.elf>"

        COMMENT "Generating BIN, HEX, and memory usage report"
        VERBATIM
)
